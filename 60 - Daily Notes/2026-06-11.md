### Option 1: Item emits "something changed", Model queries (Recommended)
```c++
// Item emits that SOMETHING changed (not visibility specifically)
class Item : public QObject
{
    Q_OBJECT
public:
    void setData(const QString& data) {
        if (m_data == data) return;
        m_data = data;
        emit internalStateChanged();  // Generic signal
    }
    
signals:
    void internalStateChanged();  // Item doesn't know about visibility
    
private:
    QString m_data;
};

// Model decides if visibility actually changed
class MyModel : public QAbstractListModel
{
    void onItemChanged(Item* item) {
        int row = m_items.indexOf(item);
        QModelIndex idx = index(row, 0);
        
        bool oldVisibility = m_cachedVisibility[row];
        bool newVisibility = item->calculateVisibility();
        
        if (oldVisibility != newVisibility) {
            m_cachedVisibility[row] = newVisibility;
            emit dataChanged(idx, idx, {IsVisibleRole});  // ✅ Only emit if changed
        }
    }
};
```
